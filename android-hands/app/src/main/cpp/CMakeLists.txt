cmake_minimum_required(VERSION 3.18.1)
project(svbridge CXX)
add_library(svbridge SHARED node_bridge.cpp)
target_compile_features(svbridge PRIVATE cxx_std_17)
# Node.js для Android (nodejs-mobile): готовая libnode.so под каждый ABI; для ABI без неё — заглушка (такие APK мы не выпускаем)
if(EXISTS ${LIBNODE_DIR}/bin/${ANDROID_ABI}/libnode.so)
  add_library(node SHARED IMPORTED)
  set_target_properties(node PROPERTIES IMPORTED_LOCATION ${LIBNODE_DIR}/bin/${ANDROID_ABI}/libnode.so)
  target_include_directories(svbridge PRIVATE ${LIBNODE_DIR}/include/node)
  target_link_libraries(svbridge node log)
else()
  target_compile_definitions(svbridge PRIVATE SV_NO_NODE=1)
  target_link_libraries(svbridge log)
endif()
